// Test-only ephemeral CMS generator. No certificate store, Authenticode signing or file execution.
using System;
using System.IO;
using System.Security.Cryptography;
using System.Security.Cryptography.Pkcs;
using System.Security.Cryptography.X509Certificates;
public sealed class FixtureCms {
    public byte[] Blob; public string Thumbprint;
    static byte[] Join(params byte[][] values) { using (MemoryStream stream=new MemoryStream()) { foreach(byte[] value in values) stream.Write(value,0,value.Length); return stream.ToArray(); } }
    static byte[] T(byte tag,byte[] body) {
        byte[] length=body.Length<128 ? new byte[] {(byte)body.Length} : body.Length<256 ? new byte[] {0x81,(byte)body.Length} : new byte[] {0x82,(byte)(body.Length>>8),(byte)body.Length};
        return Join(new byte[] {tag},length,body);
    }
    public static FixtureCms Make(string mode) {
        using (RSACng rsa=new RSACng(2048)) {
            CertificateRequest request=new CertificateRequest(new X500DistinguishedName("CN=PC Steward In-Memory Fixture"),rsa,HashAlgorithmName.SHA256,RSASignaturePadding.Pkcs1);
            using (X509Certificate2 certificate=request.CreateSelfSigned(DateTimeOffset.UtcNow.AddMinutes(-1),DateTimeOffset.UtcNow.AddHours(1))) {
                byte[] sha256=T(6,new byte[] {0x60,0x86,0x48,1,0x65,3,4,2,1});
                byte[] sha1=T(6,new byte[] {0x2b,0x0e,3,2,0x1a});
                byte[] algorithm=T(0x30,Join(mode=="sha1-pe" ? sha1 : sha256,new byte[] {5,0}));
                byte[] digest=T(0x30,Join(algorithm,T(4,new byte[mode=="sha1-pe" ? 20 : 32])));
                byte[] imageData=new byte[] {0x30,0x17,6,0x0a,0x2b,6,1,4,1,0x82,0x37,2,1,0x0f,0x30,9,3,1,0,0xa0,4,0xa2,2,0x80,0};
                if (mode=="wrong-image-type") imageData[13]=0x10;
                byte[] indirect=T(0x30,Join(imageData,digest));
                SignedCms cms=new SignedCms(new ContentInfo(new Oid("1.3.6.1.4.1.311.2.1.4"),indirect),false);
                CmsSigner signer=new CmsSigner(SubjectIdentifierType.IssuerAndSerialNumber,certificate);
                signer.IncludeOption=X509IncludeOption.EndCertOnly;
                signer.DigestAlgorithm=new Oid(mode=="sha1-signer" ? "1.3.14.3.2.26" : "2.16.840.1.101.3.4.2.1");
                if (mode=="unsigned-attribute") signer.UnsignedAttributes.Add(new AsnEncodedData(new Oid("1.2.3.4"),new byte[] {5,0}));
                // The profile rejects ANY countersignature attribute before parsing its signer.
                // Framework's CounterSign API cannot use this ephemeral CNG key on this host.
                if (mode=="counter-signature") signer.UnsignedAttributes.Add(new AsnEncodedData(new Oid("1.2.840.113549.1.9.6"),new byte[] {0x30,0}));
                if (mode=="signed-attribute") signer.SignedAttributes.Add(new AsnEncodedData(new Oid("1.2.3.4"),new byte[] {5,0}));
                if (mode=="extra-certificate") {
                    CertificateRequest otherRequest=new CertificateRequest(new X500DistinguishedName("CN=Extra Synthetic Certificate"),rsa,HashAlgorithmName.SHA256,RSASignaturePadding.Pkcs1);
                    using (X509Certificate2 extra=otherRequest.CreateSelfSigned(DateTimeOffset.UtcNow.AddMinutes(-1),DateTimeOffset.UtcNow.AddHours(1))) { signer.Certificates.Add(extra); cms.ComputeSignature(signer,true); }
                } else { cms.ComputeSignature(signer,true); }
                return new FixtureCms {Blob=cms.Encode(),Thumbprint=certificate.Thumbprint};
            }
        }
    }
}
